// backend/src/routes/api.routes.ts
// Rotas da API RESTful

import { Router, Request, Response } from 'express';
import { userService, ticketService, commentService, auditService } from '../services/database.service';
import { authMiddleware } from '../middleware/auth.middleware';
import { Logger } from '../services/logger.service';

const logger = new Logger('API Routes');
const router = Router();

// ============================================
// MIDDLEWARE
// ============================================

// Apply auth middleware a todas as rotas
router.use(authMiddleware);

// ============================================
// USUÁRIOS
// ============================================

/**
 * GET /api/users/me
 * Obter dados do usuário logado
 */
router.get('/users/me', async (req: Request, res: Response) => {
  try {
    const userId = (req as any).user.sub;
    const user = await userService.findById(userId);

    if (!user) {
      return res.status(404).json({ error: 'Usuário não encontrado' });
    }

    res.json({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        picture: user.picture,
        role: user.role,
        createdAt: user.createdAt,
      },
    });
  } catch (error) {
    logger.error('Erro ao buscar usuário:', error);
    res.status(500).json({ error: 'Erro ao buscar usuário' });
  }
});

/**
 * POST /api/users/register
 * Registrar novo usuário (login)
 */
router.post('/users/register', async (req: Request, res: Response) => {
  try {
    const { email, name, picture } = req.body;

    if (!email || !name) {
      return res.status(400).json({ error: 'Email e nome são obrigatórios' });
    }

    // Verifica se usuário já existe
    let user = await userService.findByEmail(email);

    if (!user) {
      // Cria novo usuário
      user = await userService.create(email, name, picture);
      logger.info(`Novo usuário registrado: ${email}`);
    }

    // Atualiza último login
    await userService.updateLastLogin(user.id);

    // Registra auditoria
    await auditService.log('login', 'User', user.id, user.id, email);

    res.json({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        picture: user.picture,
        role: user.role,
      },
    });
  } catch (error) {
    logger.error('Erro ao registrar usuário:', error);
    res.status(500).json({ error: 'Erro ao registrar usuário' });
  }
});

/**
 * GET /api/tickets
 * Listar tickets com filtros opcionais
 */
router.get('/tickets', async (req: Request, res: Response) => {
  try {
    const { status, priority, category } = req.query;
    const filters: any = {};

    if (status) filters.status = status;
    if (priority) filters.priority = priority;
    if (category) filters.category = category;

    const tickets = await ticketService.findMany(filters);

    res.json({
      success: true,
      count: tickets.length,
      tickets,
    });
  } catch (error) {
    logger.error('Erro ao listar tickets:', error);
    res.status(500).json({ error: 'Erro ao listar tickets' });
  }
});

/**
 * POST /api/tickets
 * Criar novo ticket
 */
router.post('/tickets', async (req: Request, res: Response) => {
  try {
    const tokenUser = (req as any).user;
    // Valida se usuário existe no banco
    const user = await userService.findById(tokenUser.sub);
    if (!user) {
      return res.status(401).json({ error: 'Usuário não encontrado no sistema' });
    }
    const userId = user.id;
    
    const { title, description, category, priority, requesterName, requesterEmail } = req.body;

    if (!title || !description) {
      return res.status(400).json({ error: 'Título e descrição são obrigatórios' });
    }

    const ticket = await ticketService.create({
      title,
      description,
      category: category || 'general',
      priority: priority || 'medium',
      status: 'open',
      requesterName,
      requesterEmail,
      createdById: userId,
    });

    // Registra auditoria
    await auditService.log('create', 'Ticket', ticket.id, userId, (req as any).user.email);

    res.status(201).json({
      success: true,
      ticket,
      message: 'Ticket criado com sucesso',
    });
  } catch (error) {
    logger.error('Erro ao criar ticket:', error);
    res.status(500).json({ error: 'Erro ao criar ticket' });
  }
});

/**
 * PUT /api/tickets/:id
 * Atualizar ticket
 */
router.put('/tickets/:id', async (req: Request, res: Response) => {
  try {
    const userId = (req as any).user.sub;
    const { id } = req.params;
    const { status, priority, assignedToId, description } = req.body;

    const updateData: any = {};
    if (status) updateData.status = status;
    if (priority) updateData.priority = priority;
    if (assignedToId) updateData.assignedToId = assignedToId;
    if (description) updateData.description = description;

    const ticket = await ticketService.update(id, updateData);

    // Registra auditoria
    await auditService.log('update', 'Ticket', id, userId, (req as any).user.email, updateData);

    res.json({
      success: true,
      ticket,
      message: 'Ticket atualizado com sucesso',
    });
  } catch (error) {
    logger.error('Erro ao atualizar ticket:', error);
    res.status(500).json({ error: 'Erro ao atualizar ticket' });
  }
});
